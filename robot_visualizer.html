// --- 1. Movement Functions (Take coord, return updated coord) ---
int moveUp(int y)    { if (y > 0) y--; return y; }
int moveDown(int y)  { if (y < 4) y++; return y; }
int moveLeft(int x)  { if (x > 0) x--; return x; }
int moveRight(int x) { if (x < 4) x++; return x; }

// --- 2. Main Control Loop ---
int main() {
    int x = 2, y = 2; // Starting coordinates
    char cmd[20];
    printf("Robot ready at (%d, %d)\n", x, y);

    while (1) { // Loops continuously until 'exit'
        printf("Enter command: ");
        scanf("%s", cmd);

        if (strcmp(cmd, "exit") == 0) {
            printf("Robot stopped.\n");
            break; // Terminates the loop safely
        }

        if (strcmp(cmd, "up") == 0 || strcmp(cmd, "u") == 0)         y = moveUp(y);
        else if (strcmp(cmd, "down") == 0 || strcmp(cmd, "d") == 0)  y = moveDown(y);
        else if (strcmp(cmd, "left") == 0 || strcmp(cmd, "l") == 0)  x = moveLeft(x);
        else if (strcmp(cmd, "right") == 0 || strcmp(cmd, "r") == 0) x = moveRight(x);
        else printf("Unknown command!\n");

        printf("Position: (%d, %d)\n", x, y);
    }
    return 0;
}
Why Parameters & Return?

Functions receive the coordinate as an argument, check the 0–4 bounds, and return the safe new coordinate (y = moveUp(y);).

Loop & Exit

while (1) runs infinitely. The only clean escape is strcmp(cmd, "exit") == 0 which triggers break;.
